# 1. 初始化 / initialize
curl -X POST {{BASE}}/api/video-uploads \
  -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -d '{"filename":"clip.mp4","size":8388608,"name":"clip","visibility":"private","fingerprint":"<64 lowercase hex>"}'

# 2. 上传第 0 片 / upload part 0
curl -X PUT {{BASE}}/api/video-uploads/$UPLOAD_ID/parts/0 \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Range: bytes 0-8388607/8388608" \
  -H "X-Chunk-SHA256: <part sha256>" \
  -H "Content-Type: application/octet-stream" --data-binary @part-0

# 3. 以服务端状态为准续传，再完成 / reconcile status, then complete
curl {{BASE}}/api/video-uploads/$UPLOAD_ID -H "Authorization: Bearer $TOKEN"
curl -X POST {{BASE}}/api/video-uploads/$UPLOAD_ID/complete -H "Authorization: Bearer $TOKEN"
