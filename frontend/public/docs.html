# 注册
curl -X POST {{BASE}}/api/auth/register \
  -H "Content-Type: application/json" \
  -d '{"username":"alice","password":"pass123"}'

# 登录，拿到 token
TOKEN=$(curl -X POST {{BASE}}/api/auth/login \
  -d "username=alice&password=pass123" \
  | python3 -c "import sys,json;print(json.load(sys.stdin)['access_token'])")

# 用 Key 上传 / 下载 / 删除（Authorization: Bearer 或 X-API-Key 均可）
curl -X POST {{BASE}}/api/upload \
  -H "Authorization: Bearer <key>" -F "file=@a.png" -F "name=测试"
curl -o a.png "{{BASE}}/i/<code>" -H "Authorization: Bearer <key>"
curl -X DELETE "{{BASE}}/api/images/<code>" -H "Authorization: Bearer <key>"
